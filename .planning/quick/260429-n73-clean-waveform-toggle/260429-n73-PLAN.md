```typescript
// ─────────────────────────────────────────────────────────────────────────────
// setVoiceDistortionBypass — bypass or restore waveshaper distortion for a voice
// bypass=true  → waveshaper.curve = null (transparent pass-through; raw oscillator to analyser)
// bypass=false → waveshaper.curve = makeDistortionCurve(saturation) (restore colour-mapped distortion)
// No-ops gracefully if the voice does not exist.
// ─────────────────────────────────────────────────────────────────────────────
export function setVoiceDistortionBypass(shapeId: string, bypass: boolean, saturation = 0): void {
  const voice = voices.get(shapeId)
  if (!voice) return
  voice.waveshaper.curve = bypass ? null : makeDistortionCurve(saturation)
}
```

The `saturation` parameter defaults to 0 so callers can pass `shape.color.s` when restoring; no other files require changes inside audioEngine.ts.
